# PRD Checklist
checklist:
  name: "prd-checklist"
  version: "1.0"
  categories:
    - name: "Problem Definition"
      items:
        - "Problem statement clearly articulated"
        - "Current pain points documented"
        - "Target audience identified"
        - "Success metrics defined (quantifiable)"
        
    - name: "Goals & Scope"
      items:
        - "MVP scope clearly bounded"
        - "Phase 2+ scope identified but not detailed"
        - "Out-of-scope items explicitly listed"
        - "Goals are SMART (Specific, Measurable, Achievable, Relevant, Time-bound)"
        
    - name: "User Stories"
      items:
        - "All user roles identified"
        - "Stories follow INVEST criteria"
        - "Each story has acceptance criteria"
        - "Stories mapped to Python features"
        - "Priority assigned (MoSCoW)"
        - "Dependencies between stories noted"
        
    - name: "Functional Requirements"
      items:
        - "All CPU cores covered"
        - "All peripherals covered"
        - "Display rendering features covered"
        - "Audio features covered"
        - "Config system covered"
        - "Debug features covered"
        - "Serial/IR features covered"
        - "Save/load state covered"
        - "PortMaster integration covered"
        
    - name: "Non-Functional Requirements"
      items:
        - "Performance targets specified (FPS, frame time)"
        - "Memory limits specified"
        - "Startup time specified"
        - "Accuracy requirements specified"
        - "Platform constraints documented"
        - "Portability requirements specified"
        
    - name: "Technical Constraints"
      items:
        - "Target architectures listed (ARM32/ARM64)"
        - "OS version minimum specified"
        - "Library dependencies listed"
        - "Build system requirements"
        - "Packaging requirements"
        
    - name: "Risks & Assumptions"
      items:
        - "Technical risks identified and rated"
        - "Schedule risks identified"
        - "Mitigation strategies for high risks"
        - "Assumptions documented and validated"
        
    - name: "Acceptance Criteria"
      items:
        - "Definition of Done for project"
        - "Definition of Done for each story"
        - "Sign-off process defined"
        - "Rollback criteria defined"
