# Porting Workflow - Python to C++ for PortMaster
workflow:
  name: "porting-workflow"
  version: "1.0"
  description: "Detailed porting process per component"
  steps:
    - name: "Component Analysis"
      description: "Analyze Python component before porting"
      agent: "developer"
      tasks:
        - "Read Python source thoroughly"
        - "Document public API surface"
        - "Identify algorithmic complexity"
        - "Note Python-specific idioms (dicts, lists, dynamic typing)"
        - "Map data structures to C++ equivalents"
        - "Identify performance hotspots"
      output: "Porting analysis document per component"
      
    - name: "Interface Design"
      description: "Design C++ interface before implementation"
      agent: "architect"
      tasks:
        - "Define pure virtual interfaces"
        - "Specify ownership semantics (unique_ptr/shared_ptr)"
        - "Define error handling (std::expected/error_code)"
        - "Document thread safety guarantees"
        - "Specify const-correctness"
      output: "Header files (.hpp) with interfaces"
      
    - name: "Implementation"
      description: "Port implementation to C++"
      agent: "developer"
      tasks:
        - "Translate algorithms 1:1 first (correctness over speed)"
        - "Replace Python dicts with std::unordered_map/flat_map"
        - "Replace Python lists with std::vector"
        - "Handle dynamic typing with variant/any or templates"
        - "Implement memory management (RAII)"
        - "Add unit tests for each function"
      output: "Source files (.cpp) + unit tests"
      
    - name: "Integration"
      description: "Integrate component into emulator"
      agent: "developer"
      tasks:
        - "Wire into component system"
        - "Connect message passing (lock-free queues)"
        - "Verify timing integration"
        - "Test with real .brick configs"
      output: "Integrated component"
      
    - name: "Validation"
      description: "Validate against Python reference"
      agent: "qa-engineer"
      tasks:
        - "Run side-by-side comparison"
        - "Compare instruction traces"
        - "Compare VRAM output frame-by-frame"
        - "Compare audio output"
        - "Benchmark performance"
      output: "Validation report"
      
    - name: "Optimization"
      description: "Optimize for target platform"
      agent: "developer"
      tasks:
        - "Profile on target hardware"
        - "Apply ARM NEON intrinsics where beneficial"
        - "Optimize memory layout (cache-friendly)"
        - "Reduce allocations in hot paths"
        - "Verify correctness after optimization"
      output: "Optimized component + benchmarks"

  python_to_cpp_mapping:
    multiprocessing_Queue: "lockfree::spsc_queue / moodycamel::ConcurrentQueue"
    multiprocessing_Process: "std::thread + affinity"
    dict: "std::unordered_map / absl::flat_hash_map"
    list: "std::vector"
    set: "std::unordered_set"
    tuple: "std::tuple / struct"
    dataclass: "struct with constructor"
    property: "getter/setter methods"
    "@dataclass": "struct + aggregate initialization"
    "asyncio": "std::coroutine (C++20) / fibers"
    "time.perf_counter_ns": "std::chrono::steady_clock::now()"
    "sleep": "std::this_thread::sleep_for"
    "ctypes.windll": "Platform-specific #ifdef _WIN32"
    "JSON": "nlohmann::json / simdjson"
    "SVG parsing": "lunasvg / nanosvg"
    "PyQt6 graphics": "OpenGL ES 2.0 + SDL2"
    "PyQt6 audio": "SDL2 Audio / miniaudio"
